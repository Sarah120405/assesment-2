CREATE TABLE bikes (
id INT PRIMARY KEY AUTO_INCREMENT,
name VARCHAR(100) NOT NULL,
category VARCHAR(50),
price DECIMAL(10,2),
stock INT
);

CREATE TABLE orders (
id INT PRIMARY KEY AUTO_INCREMENT,
bike_id INT,
qty INT,
FOREIGN KEY (bike_id) REFERENCES bikes(id)
);

SELECT b.name, SUM(o.qty) AS total_qty
FROM bikes b
JOIN orders o ON b.id = o.bike_id
GROUP BY b.name
HAVING SUM(o.qty) > 5;

CREATE INDEX idx_bikes_category ON bikes(category);

-- Index helps when filtering or grouping by category, such as: WHERE category = 'Electric' or when using category in query conditions to reduce table scans.
